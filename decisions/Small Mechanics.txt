# Outlaws
political_decisions = {
	hunt_for_outlaws = {
		potential = {
			any_owned_province = { has_province_modifier = outlaws }
		}
		allow = {
			any_owned_province = {
				has_province_modifier = outlaws
				OR = {
					has_building = fort
					country_units_in_province = THIS
				}
			}
			money = 10000
		}
		effect = {
			random_owned = {
				limit = {
					has_province_modifier = outlaws
					OR = {
						has_building = fort
						country_units_in_province = THIS
					}
				}
				add_province_modifier = { name = hunting_outlaws duration = -1 }
				remove_province_modifier = outlaws
			}
			treasury = -10000
			
			hunting_outlaws_loc = {}
		}
	}
}
